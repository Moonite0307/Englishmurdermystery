# 运行时补充提示词与修复逻辑

这些是当前源码片段，供核对实际请求组成；不必在这里修改 JavaScript。它们并非每次都会全部发送。若想改变相应行为，可以直接在片段下用中文写修改意见。

## 生成校对和修复（首次请求之后最多重试两次）

```javascript
async function generateValidatedScript(prompt) {
  let request = [{ role: "user", content: prompt }];
  let currentData = null;
  let repairFields = null;
  for (let attempt = 0; attempt < 3; attempt++) {
    let raw = "", incomplete = false;
    try {
      raw = await callLLM(request, null, { temperature: 0.4, maxTokens: 8192 });
    } catch (error) {
      if (error.code !== "MODEL_OUTPUT_LENGTH") throw error;
      incomplete = true;
      raw = error.partialText || "";
    }
    let data = null;
    if (!incomplete) {
      if (repairFields && currentData) data = applyScriptPlanRepair(currentData, raw, repairFields);
      else try { data = normalizeScriptPlanReferences(parseModelJson(raw)); } catch (_) {}
    }
    if (!data && repairFields && currentData && !incomplete) data = currentData;
    const errors = incomplete ? ["模型输出达到长度上限，JSON底稿被截断"] : validateScriptPlan(data);
    if (!errors.length) return data;
    console.warn("[剧本校对]", errors);
    if (attempt === 2) {
      const failure = new Error(incomplete
        ? "模型多次输出到一半达到长度上限，精简重写后仍未完成。请稍后重试。"
        : "剧本自动校对后仍未通过：" + scriptPlanFailureSummary(errors) + "。请重新生成。");
      failure.validationErrors = errors;
      throw failure;
    }
    currentData = data;
    repairFields = scriptPlanRepairFields(errors, data);
    if ($("planning-status")) $("planning-status").textContent = incomplete
      ? "故事输出过长，正在精简重写……"
      : repairFields
        ? "正在补全人物关系与证据链（" + (attempt + 1) + "/2）……"
        : "正在校对角色、线索与任务（" + (attempt + 1) + "/2）……";
    // 只携带最近一稿，避免每次修复都叠加整份长底稿。
    const contextDraft = data ? JSON.stringify(data) : raw;
    const repairInstruction = repairFields
      ? "尚未开局。只返回JSON补丁，格式为{\"updates\":{\"字段名\": [...]}}，updates只允许包含这些顶层字段：" + repairFields.join("、") +
        "。仅修复这些字段并保留其他全部内容，尤其不得改动truth、locked_ending和人物设定；不要输出完整剧本、解释或Markdown。具体问题：" + errors.join("；") +
        (repairFields.includes("relationships") ? "。关系须引用现有角色，补上真实、符合各自secret/agenda的直接关系，使每人至少关联两人，不要用虚构角色凑数。" : "") +
        (repairFields.includes("core_questions") ? "。每个问题须关联至少两条现有的表决前evidence线索，来源不同；direction_hint不得透露答案或角色姓名。" : "") +
        (repairFields.includes("task_criteria") ? "。按原要求补齐全部三项task_criteria及requirements：inspect；inquire+explain；judge+reason。前两类required=2且提供现有目标id/label/中英文aliases，其余required=1且targets=[]。可见条件不能透露秘密。" : "") +
        (repairFields.includes("clues") ? "。只补足证据链所必需的线索，保持与固定真相和剧情一致，维护有效ID、幕次、来源及交叉引用。" : "")
      : "尚未开局。请修复以下问题，返回完整JSON底稿，保留原设定并同步所有引用；不得省略字段：" + errors.join("；") +
        "。每个叙述字段只写必要短句，core_vocab保持[]，不要预演对话或输出解释。提示只比较已获得的线索，不写答案和角色姓名。" +
        (incomplete || !data ? "上一稿格式未完成，请从头输出一份更精简的完整JSON，不要接着残片续写。" : "");
    request = [
      { role: "user", content: prompt },
      { role: "assistant", content: contextDraft },
      { role: "user", content: repairInstruction }
    ];
  }
}
```

## 英文选项模式的附加提示词

```javascript
function optionModeRule() {
  if (!isOptionMode()) return "";
  return "【英文选项模式·最高优先级交互规则】玩家全程只能点击你提供的英文选项，不能手动输入。每轮包括开场、自我介绍、搜证、公聊、私聊、推理、表决、结局及复盘，都必须给出2至4个可直接点击的英文完整句子options和逐项对应的中文option_translations。task_type用choice或search，禁止free、grammar或要求玩家另行打字。选项必须贴合当前已公开信息和任务的可观察完成条件，包含可以实际执行的查看、向具体角色询问、提出解释、提交判断及理由等行动；用玩家第一人称写可选择的观点，选择即代表玩家采纳该观点，task_events仍只引用玩家本轮实际点击的逐字句子，不替玩家自动完成其他任务。不得用选项提前公开尚未释放的线索、凶手、秘密或固定真相，也不得改变开局锁定结局。阶段已满足后给出进入下一幕或表决、结局、复盘的选项；复盘完整后再按原规则game_over。选项不得全是重复追问或无关闲聊，不得要求玩家提供你未给出的文字。";
}
```

## 自由对话回答语言的附加提示词

```javascript
function replyLanguageRule() {
  if (isOptionMode()) return optionModeRule() + " NPC台词继续使用英语，translation提供中文翻译。主线任务由应用核验本轮实际选择的行动，不因语法练习阻止任务打勾。";
  const mode = scriptInfo.replyLanguage === "zh"
    ? "玩家选择中文回答模式：允许中文、英文和中英混合输入，覆盖其他要求玩家必须用英语回答的描述。按表达的意思判断行动和任务，不能要求改成英文后才推进。中文不做英语语法纠错；若含英文，仅检查英文部分，可自然给出对应英语表达但不强制重答。"
    : "玩家选择英文回答模式：玩家的中文输入已由前端阻止发送，使用英文开展互动。";
  return "【本局回答语言规则】" + mode + " NPC台词继续使用英语，translation继续提供完整中文翻译。主线任务完成与语言练习独立，不因语法小错阻止任务打勾。";
}
```

## 开局系统提示词拼接和隐藏开场消息

```javascript
const daily = Boolean(scriptInfo.dailyMode);
  let systemPrompt;
  if (daily) {
    /* 闯关/随机模式：注入真实场景的「结构性信息 + 重点词」，
       绝不注入 turns 或 dialogue_text 原文；剧本由模型按目标原创。 */
    const ctx = scriptInfo.dailyCtx || {};
    systemPrompt = fillTemplate(PROMPT_PLAY_DAILY_GAME, {
      dailyMode: scriptInfo.dailyMode,
      blockName: ctx.blockName || scriptInfo.dailyBlockId || "日常场景交涉",
      sceneName: ctx.sceneName || scriptInfo.dailyScenes || scriptInfo.theme,
      dailyLevel: scriptInfo.level,
      taskGoal: ctx.taskGoal || "（未提供）",
      successCriteria: ctx.successCriteria || "（未提供）",
      commFunctions: ctx.commFunctions || "（未提供）",
      targetWords: ctx.targetWordsText || "（未提供）",
      scriptPlan: scriptPlan,
      coreVocab: coreVocabText()
    });
  } else {
    systemPrompt = PROMPT_PLAY_GAME
      .replace("{genre}", scriptInfo.genre)
      .replace("{genreDesc}", genreDesc() + "；" + genrePlay())
      .replace("{scriptPlan}", scriptPlan)
      .replace("{playerScript}", playerScriptText())
      .replace("{npcCount}", String(npcCount()))
      .replace("{coreVocab}", coreVocabText())
      .replace("{challengeDesc}", challengeDesc());
  }
  messages = [{ role: "system", content: systemPrompt + "\n" + replyLanguageRule() }];

  /* 隐藏的启动消息：按固定流程开场——读本、介绍 AI 角色、引导自我介绍（不展示在聊天界面） */
  const boot = daily
    ? "（关卡开始。请用英语介绍生活情境、玩家身份和当前需要解决的问题，然后给出第一个清楚的交涉目标。若需要查看票据、菜单、路线或房间资料，可以使用 search；否则以 free 对话开始。speaker 必须填写出场 NPC 名字，重点练习词要自然出现。）"
    : "（游戏开始。请严格按固定流程推进：先以 DM 身份用英语交代背景氛围（speaker 填 DM），并让登场的主要 AI 角色依次亮相互动一两句，营造群像感；然后邀请玩家按本局所选回答语言做角色自我介绍。本轮 task_type 用 free，不给选项。本剧考纲词汇表中的词要开始分批融入台词。）";
  const bootMessage = boot + (isOptionMode() ? " 本局选择了英文选项模式，覆盖上述free或不给选项的描述：开场也必须用choice并提供2至4个英文自我介绍或初始行动选项及对应中文翻译，玩家不能打字。" : "");
  messages.push({ role: "user", content: bootMessage });
  await requestNpcReply(bootMessage, true);
}
```
